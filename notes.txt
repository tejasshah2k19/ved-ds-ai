custom index 

marks = [80,70,90]

s1 = pd.Series(marks)

s1 = pd.Series(marks,index=["ram","shyam","sita"])
print(s1)

output: 
ram      80
shyam    70
sita     90
dtype: int64


what if we have duplicate labeled?

s1 = pd.Series([90,80,70,60],index=["ram","shyam","sita","ram"])
print(s1)

output: 
ram      90
shyam    80
sita     70
ram      60
dtype: int64

#accessing Series elements ?

s1 = pd.Series([10,20,30,40,50])

print(s1[0]) #index 0 => 10  
print(s1[1]) #index 1 => 20 


#accessing using custom labeles 

s1 = pd.Series([90,80,70,60],index=["ram","shyam","sita","ram"])
print(s1["ram"])
print(s1["sita"])

x = s1["ram"] #Series 
y = s1["sita"] #value - int 

print(type(x))
print(type(y))



#accessing using slicing 
s1 = pd.Series([90,80,70,60])
print(s1[0:2]) # 0,1


#import Series attributes
index -> return index labels 
values -> return actual values 
dtype -> return data type 
size -> return number of elements
shape -> return dimension
ndim - return number of dimensions
empty -> check whether Series is empty 
name -> return Series name 

print(s1.empty)
print(s1.size)
print(s1.ndim)
print(s1.index)
print(s1.values)
print(s1.name)
